cmake_minimum_required(VERSION 3.16)
project(hmos_ffmpeg LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH TRUE)
set(FFMPEG_PREFIX "$ENV{FFMPEG_PREFIX}" CACHE PATH "Built FFmpeg prefix")
if(NOT EXISTS "${FFMPEG_PREFIX}/include/libavcodec/avcodec.h")
  message(FATAL_ERROR "Build FFmpeg first and set FFMPEG_PREFIX")
endif()
foreach(component avformat avcodec avfilter swscale swresample avutil)
  add_library(ff_${component} SHARED IMPORTED)
  file(GLOB files "${FFMPEG_PREFIX}/lib/lib${component}-*.so")
  list(LENGTH files count)
  if(NOT count EQUAL 1)
    message(FATAL_ERROR "Expected one versioned ${component} shared library")
  endif()
  set_target_properties(ff_${component} PROPERTIES IMPORTED_LOCATION "${files}")
endforeach()
add_library(hmosffmpeg SHARED engine.cpp napi.cpp)
target_include_directories(hmosffmpeg PRIVATE "${FFMPEG_PREFIX}/include")
target_link_libraries(hmosffmpeg PRIVATE ff_avformat ff_avcodec ff_avfilter ff_swscale ff_swresample ff_avutil libace_napi.z.so)
target_compile_options(hmosffmpeg PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(hmosffmpeg PRIVATE -Wl,-z,relro,-z,now,--build-id=sha1)
